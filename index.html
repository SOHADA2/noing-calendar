구글 시트 연결

구글 시트를 데이터 저장소로 사용합니다. 한 번 셋업하면 어느 기기에서나 같은 데이터를 보고, 시트에서 직접 수정도 가능합니다.

처음 셋업하는 방법 (약 5분)

1. 구글 시트 새로 만들기 → sheets.google.com

2. 시트 1행(헤더)에 다음 텍스트를 복사해서 A1에 붙여넣기 (10개 칸에 자동 분배):

id date time company title channel type owner status memo

3. 시트 메뉴 확장 프로그램 → Apps Script → 기존 코드 모두 지우고 아래 붙여넣은 뒤 저장(💾):

function getSheet_(name) { const ss = SpreadsheetApp.getActiveSpreadsheet(); if (name) { const s = ss.getSheetByName(name); if (s) return s; } return ss.getSheets()[0]; } function doGet(e) { const sheetName = e && e.parameter ? e.parameter.sheet : null; const sheet = getSheet_(sheetName); const data = sheet.getDataRange().getValues(); const headers = data.shift(); const rows = data .filter(r => r[0] !== '' && r[0] !== null) .map(r => { const o = {}; headers.forEach((h, i) => { let v = r[i]; if (v instanceof Date) { v = (h === 'time') ? Utilities.formatDate(v, 'Asia/Seoul', 'HH:mm') : Utilities.formatDate(v, 'Asia/Seoul', 'yyyy-MM-dd'); } o[h] = v; }); return o; }); return ContentService.createTextOutput(JSON.stringify(rows)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const body = JSON.parse(e.postData.contents); const sheet = getSheet_(body.sheet); const data = sheet.getDataRange().getValues(); const headers = data[0]; const row = headers.map(h => body[h] !== undefined ? body[h] : ''); if (body.action === 'add') { sheet.appendRow(row); } else if (body.action === 'update') { for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(body.id)) { sheet.getRange(i + 1, 1, 1, headers.length).setValues([row]); break; } } } else if (body.action === 'delete') { for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(body.id)) { sheet.deleteRow(i + 1); break; } } } return ContentService.createTextOutput(JSON.stringify({ ok: true })) .setMimeType(ContentService.MimeType.JSON); }

4. Apps Script 우측 상단 배포 → 새 배포

  • 유형(톱니바퀴) → 웹 앱
  • 다음 사용자 인증으로 실행: 나
  • 액세스 권한이 있는 사용자: 모든 사용자
  • 배포 → 권한 승인 ("고급" → "안전하지 않은 페이지로 이동" → 허용)
  • 표시되는 웹 앱 URL 복사

5. 위 입력칸에 URL 붙여넣기 → "저장 후 동기화"

참고: 처음 동기화 시, 현재 화면에 있는 76건(2026년 데이터)이 시트에 자동 업로드됩니다. 시트가 비어있어야 깔끔하게 들어갑니다.
